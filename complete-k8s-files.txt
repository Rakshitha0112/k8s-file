17. Deployment file for emptyDir


apiVersion: apps/v1
kind: Deployment
metadata:
  name: d1
spec:
   replicas: 3
   selector:
      matchLabels:
          app: web
   template:
      metadata:
         labels:
            app: web
      spec:
          containers:
             - name: c1
               image: ubuntu
               command: ["/bin/bash",'-c',"while true; do echo hello;sleep 300; done"]
               volumeMounts:
                      - name: vname
                        mountPath: /data/path
             - name: c2
               image: amazonlinux
               command: ["/bin/bash",'-c',"while true; do echo hello;sleep 300; done"]
               volumeMounts:
                      - name: vname
                        mountPath: /data/path
          volumes:
           - name: vname
             emptyDir: {}
